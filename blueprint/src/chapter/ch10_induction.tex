\chapter{The main induction step}
\label{chap:induction}
\label{sec:induction}

\begin{definition}[Induction error]\label{def:main-induction-error}
	\lean{MIPStarRE.LDT.MainInductionStep.simplifiedMainInductionError, MIPStarRE.LDT.MainInductionStep.simplifiedDegreeZeroError, MIPStarRE.LDT.MainInductionStep.simplifiedPositiveDegreeError}
	\leanok
	Set $\sigma_{m,d}(\eps,\delta,\gamma)=\delta+\sqrt{8m\eps}$ if $d=0$, and
	\[
		\sigma_{m,d}(\eps,\delta,\gamma)=1000d^2m^4\bigl(\eps^{1/32}+\delta^{1/32}
		+(d/q)^{1/32}+\gamma^{1/8}\bigr)
	\]
	if $d\ge1$.
\end{definition}

\begin{theorem}[Main induction]\label{thm:main-induction}
	\lean{MIPStarRE.LDT.MainInductionStep.simplifiedMainInduction, MIPStarRE.LDT.MainInductionStep.answerStrategyOfSymStrat, MIPStarRE.LDT.MainInductionStep.answerStrategyOfSymStrat_isGood}
	\leanok
	\uses{def:good-strategy, def:polymeasurements, def:simeq, def:main-induction-error}
	Let $m\ge1$ and let $(\psi,A,B,L)$ be an $(\eps,\delta,\gamma)$-good
	symmetric strategy for the $(m,q,d)$ low individual degree test.  There is
	a measurement $G\in\polymeas mqd$ such that, on average over
	$u\sim\F_q^m$,
	\[
		A^u_a\ot I\simeq_{\sigma_{m,d}(\eps,\delta,\gamma)}I\ot G_{[g(u)=a]}.
	\]
\end{theorem}

\begin{proof}
	\leanok
	\uses{lem:answer-main-induction}
	Relabel the diagonal-line outcomes of the strategy as functions on the
	line; this gives a strategy with the relaxed diagonal answer alphabet and
	the same failure probabilities, and the point measurements are unchanged.
	Apply Lemma~\ref{lem:answer-main-induction}, proved in
	Section~\ref{sec:proof-of-main-induction}.
\end{proof}


Theorem~\ref{thm:main-induction} produces a measurement consistent with $A$,
but not projective and not self-consistent, and it assumes a symmetric
strategy.  Section~\ref{sec:proof-of-main-formal} removes these deficiencies.
The induction itself is proved in Section~\ref{sec:proof-of-main-induction}
from self-improvement and pasting.

\section{Proof of the main theorem}
\label{sec:proof-of-main-formal}

\begin{lemma}[Role-register symmetrization]\label{lem:role-register-symmetrization}
	\lean{MIPStarRE.LDT.ProjStrat.roleRegisterSymmStrategy, MIPStarRE.LDT.ProjStrat.roleRegisterSymmStrategy_is_good_three_mul}
	\leanok
	\uses{def:projective-strategy, def:symmetric-projective-strategy, def:good-strategy}
	Let
	$(\psi,A^{\mathrm A},B^{\mathrm A},L^{\mathrm A},A^{\mathrm B},B^{\mathrm B},L^{\mathrm B})$
	be a projective strategy passing the $(m,q,d)$ test with probability at
	least $1-\eps$.  On the local space
	$\mathbb C^{\{\mathrm A,\mathrm B\}}\ot(\calH_{\mathrm A}\oplus\calH_{\mathrm B})$
	there is a symmetric projective strategy that is
	$(3\eps,3\eps,3\eps)$-good, whose state places $\psi$ and its swap in the
	two role sectors and whose measurements act blockwise by
	$A^{\mathrm A}\oplus A^{\mathrm B}$, $B^{\mathrm A}\oplus B^{\mathrm B}$,
	and $L^{\mathrm A}\oplus L^{\mathrm B}$.
\end{lemma}
\begin{proof}
	\leanok
	Each of the three subtests is chosen with probability $1/3$, so each
	fails with probability at most $3\eps$.  The block-diagonal measurements
	reproduce the axis-parallel and diagonal-line branch averages exactly, and
	the self-consistency branch of the symmetrized strategy equals the
	original point-agreement defect.  The simplified exposition uses the
	direct sum $\calH_{\mathrm A}\oplus\calH_{\mathrm B}$ without the role
	register; the two constructions give the same estimates.
\end{proof}

\begin{lemma}[Extracting the role blocks]\label{lem:role-register-extraction}
	\lean{MIPStarRE.LDT.Measurement.extractRoleRegisterAlice, MIPStarRE.LDT.Measurement.extractRoleRegisterBob, MIPStarRE.LDT.ProjStrat.sourceRoleRegisterPointConsistency_ofSymConsistency, MIPStarRE.LDT.ProjStrat.simplifiedRoleRegisterPointConsistency}
	\leanok
	\uses{lem:role-register-symmetrization, thm:main-induction}
	Let $G$ be a measurement on the symmetrized local space with
	$A^u_{\mathrm{sym},a}\ot I\simeq_\sigma I\ot G_{[g(u)=a]}$.  Its diagonal
	blocks $G^{\mathrm A}$ and $G^{\mathrm B}$ are measurements on
	$\calH_{\mathrm A}$ and $\calH_{\mathrm B}$ with
	\[
		A^{\mathrm A,u}_a\ot I\simeq_{2\sigma}I\ot G^{\mathrm B}_{[g(u)=a]},
		\qquad
		G^{\mathrm A}_{[g(u)=a]}\ot I\simeq_{2\sigma}I\ot A^{\mathrm B,u}_a.
	\]
\end{lemma}
\begin{proof}
	\leanok
	The symmetrized point measurements are block diagonal and the two terms
	of the symmetrized state occupy orthogonal sectors, so the symmetrized
	inconsistency is one half of the sum of the two displayed inconsistencies.
\end{proof}

\begin{lemma}[Polynomial self-consistency of the blocks]\label{lem:role-register-polynomial-consistency}
	\lean{MIPStarRE.LDT.ProjStrat.sourceRoleRegisterFullPolynomialSelfConsistency_ofPointConsistency, MIPStarRE.LDT.ProjStrat.simplifiedRoleRegisterPolynomialConsistency}
	\leanok
	\uses{lem:role-register-extraction, cor:schwartz-zippel-individual}
	In the situation of Lemma~\ref{lem:role-register-extraction}, if the
	original strategy passes with probability at least $1-\eps$, then
	\[
		G^{\mathrm A}_g\ot I\simeq_\zeta I\ot G^{\mathrm B}_g,\qquad
		\zeta=2\sigma+2\sqrt{3\eps+2\sigma}+md/q.
	\]
\end{lemma}
\begin{proof}
	\leanok
	\uses{prop:simeq-to-approx, prop:cab-approx-delta}
	The point test gives $A^{\mathrm A,u}_a\ot I\simeq_{3\eps}I\ot
		A^{\mathrm B,u}_a$.  Chaining it with the two relations of
	Lemma~\ref{lem:role-register-extraction} through the projective point
	measurements shows that the evaluated blocks
	$G^{\mathrm A}_{[g(u)=a]}$ and $G^{\mathrm B}_{[g(u)=a]}$ are consistent
	with error $2\sigma+2\sqrt{3\eps+2\sigma}$.  Two distinct polynomials of
	individual degree $d$ agree at a uniform point with probability at most
	$md/q$, which bounds the extra mismatch mass of the full polynomial
	outcomes.  The simplified exposition obtains the sharper intermediate
	error $(2\sqrt\sigma+\sqrt{6\eps})^2+md/q$ by a projective comparison
	lemma; the formalization keeps the displayed estimate, which still gives
	the final exponent $1/64$.
\end{proof}

\begin{lemma}[Rounding the blocks]\label{lem:role-register-rounding}
	\lean{MIPStarRE.LDT.ProjStrat.simplifiedRoleRegisterProjectiveMeasurements, MIPStarRE.LDT.ProjStrat.simplifiedRoundedPolynomialConsistency, MIPStarRE.LDT.Test.point_consistency_of_right_rounding, MIPStarRE.LDT.Test.point_consistency_of_left_rounding}
	\leanok
	\uses{lem:role-register-polynomial-consistency, lem:orthonormalization-main-lemma}
	There are projective measurements
	$Q^{\mathrm A},Q^{\mathrm B}\in\polymeas mqd$ with
	$G^{\mathrm A}_g\ot I\approx_{18\zeta}Q^{\mathrm A}_g\ot I$,
	$I\ot G^{\mathrm B}_g\approx_{18\zeta}I\ot Q^{\mathrm B}_g$,
	\[
		Q^{\mathrm A}_g\ot I\simeq_{57\zeta}I\ot Q^{\mathrm B}_g,
	\]
	and
	$A^{\mathrm A,u}_a\ot I\simeq_{2(2\sigma+18\zeta)}I\ot Q^{\mathrm B}_{[g(u)=a]}$,
	$Q^{\mathrm A}_{[g(u)=a]}\ot I\simeq_{2(2\sigma+18\zeta)}I\ot A^{\mathrm B,u}_a$.
\end{lemma}
\begin{proof}
	\leanok
	\uses{prop:simeq-to-approx, prop:round-right-submeasurement}
	Lemma~\ref{lem:orthonormalization-main-lemma} applied to
	Lemma~\ref{lem:role-register-polynomial-consistency} in both tensor
	orders gives $Q^{\mathrm A}$ and $Q^{\mathrm B}$.  The root-distance
	triangle inequality and Proposition~\ref{prop:simeq-to-approx} give the
	polynomial consistency, and
	Proposition~\ref{prop:round-right-submeasurement} with $f_u(g)=g(u)$
	transfers the two point relations to the rounded measurements.
\end{proof}

\begin{lemma}[The nontrivial error regime]\label{lem:main-formal-small-error}
	\lean{MIPStarRE.LDT.Test.simplifiedMainFormal_smallError, MIPStarRE.LDT.Test.simplifiedFinal_induction_bound, MIPStarRE.LDT.Test.simplifiedFinal_scalar_absorption}
	\leanok
	\uses{def:main-formal-error, lem:role-register-symmetrization, thm:main-induction, lem:role-register-extraction, lem:role-register-polynomial-consistency, lem:role-register-rounding}
	If $\nu<1$, the conclusions of Theorem~\ref{thm:main-formal} hold.
\end{lemma}
\begin{proof}
	\leanok
	Here $R=K_{m,d}(\eps^{1/64}+(d/q)^{1/64})<1/21000$, so $\eps,d/q<1$.
	Symmetrize by Lemma~\ref{lem:role-register-symmetrization} and apply
	Theorem~\ref{thm:main-induction} to the $(3\eps,3\eps,3\eps)$-good
	symmetric strategy, with error
	$\sigma=\sigma_{m,d}(3\eps,3\eps,3\eps)\le10000K_{m,d}(\eps^{1/32}+(d/q)^{1/32})$.
	Lemmas~\ref{lem:role-register-extraction},
	\ref{lem:role-register-polynomial-consistency},
	and~\ref{lem:role-register-rounding} give projective measurements whose
	two point inconsistencies are at most $2(2\sigma+18\zeta)$ and whose
	polynomial inconsistency is at most $57\zeta$.  Since $K_{m,d}\ge m\ge1$
	and $R<1/21000$, one has $2\sigma\le20000R^2\le R$, $3\eps\le3R^{64}\le R^2$,
	and $md/q\le R^{64}\le R$.  Hence
	$\zeta\le R+2\sqrt{20001}\,R+R\le285R$, the point inconsistencies are
	at most $2(R+18\cdot285R)\le10300R$, and the polynomial inconsistency is
	at most $57\cdot285R\le16300R$; all are below $21000R=\nu$.
\end{proof}


\section{Self-improvement and pasting in the induction}
\label{sec:self-improvement-and-pasting}

The induction step combines Theorem~\ref{thm:self-improvement-in-induction-section},
which moves the consistency error of a slice measurement into
incompleteness and outputs projective slices on a common dilated space, with
the pasting theorem.

\begin{theorem}[Pasting in the induction]\label{thm:ld-pasting-in-induction-section}
	\lean{MIPStarRE.LDT.MainInductionStep.answerSuccessorPasting, MIPStarRE.LDT.Pasting.simplifiedInductionAttemptCount, MIPStarRE.LDT.Pasting.simplifiedInductionPaperError, MIPStarRE.LDT.Pasting.simplifiedPastingPaperError_induction_eq}
	\leanok
	\uses{thm:ld-pasting}
	In the situation of Theorem~\ref{thm:ld-pasting}, there is a measurement
	$H\in\polymeas{m+1}qd$ with
	\[
		A^u_a\ot I\simeq_\sigma I\ot H_{[h(u)=a]},\qquad
		\sigma=\frac{m+1}m\kappa+21(m+1)^2d^2\sqrt m\,s.
	\]
\end{theorem}
\begin{proof}
	\leanok
	Take $k=(m+1)d$ in Theorem~\ref{thm:ld-pasting}.  Since $m,d\ge1$, one
	has $k\ge d+1$ and $k/(k-d)=(m+1)/m$.
\end{proof}

\section{Proof of the main induction}
\label{sec:proof-of-main-induction}

\begin{definition}[Appending a coordinate]\label{def:append-coordinate}
	\lean{MIPStarRE.LDT.AxisParallelLine.appendAtHeight, MIPStarRE.LDT.AxisLinePolynomial.appendAtHeight, MIPStarRE.LDT.DiagonalLine.appendAtHeight}
	\leanok
	For $x\in\F_q$ and a line $\ell\subseteq\F_q^m$, $\mathrm{append}_x(\ell)$
	is the line $\{(u,x):u\in\ell\}$ in $\F_q^{m+1}$, and for a function $f$
	on $\ell$, $\mathrm{append}_x(f)(u,x)=f(u)$.
\end{definition}

\begin{definition}[$x$-restricted strategy]\label{def:restricted-strategy}
	\lean{MIPStarRE.LDT.MainInductionStep.xRestrictedStrategy, MIPStarRE.LDT.MainInductionStep.xRestrictedAnswerSymStrat, MIPStarRE.LDT.MainInductionStep.RestrictedSymStrat}
	\leanok
	\uses{def:append-coordinate, def:symmetric-projective-strategy}
	For a symmetric strategy $(\psi,A,B,L)$ for the $(m+1,q,d)$ test and
	$x\in\F_q$, the $x$-restricted strategy $(\psi,A^x,B^x,L^x)$ for the
	$(m,q,d)$ test is $(A^x)^u_a=A^{u,x}_a$,
	$(B^x)^\ell_f=B^{\mathrm{append}_x(\ell)}_{\mathrm{append}_x(f)}$, and
	$(L^x)^\ell_f=L^{\mathrm{append}_x(\ell)}_{\mathrm{append}_x(f)}$.  Since
	$f\mapsto\mathrm{append}_x(f)$ is a bijection between all functions on
	the two lines, $L^x$ is a complete projective measurement for the relaxed
	diagonal answer alphabet.
\end{definition}

\begin{lemma}[Restricted failure probabilities]\label{lem:restricted-probabilities}
	\lean{MIPStarRE.LDT.MainInductionStep.answerSuccessorRestrictedProbabilities, MIPStarRE.LDT.MainInductionStep.AnswerSuccessorRestrictedProbabilitiesStatement}
	\leanok
	\uses{def:restricted-strategy, def:good-strategy}
	If $(\psi,A,B,L)$ is $(\eps,\delta,\gamma)$-good for the $(m+1,q,d)$ test
	and $\eps_x,\delta_x,\gamma_x$ are the failure probabilities of the
	$x$-restricted strategy, then
	\[
		\E_x\eps_x\le\frac{m+1}m\eps,\qquad\E_x\delta_x\le\delta,\qquad
		\E_x\gamma_x\le\frac{m+1}m\gamma.
	\]
\end{lemma}
\begin{proof}
	\leanok
	The self-consistency test on a uniform point $(u,x)$ is the average over
	$x$ of the restricted self-consistency tests.  In the axis-parallel and
	diagonal tests the sampled direction lies in the first $m$ coordinates
	with probability $m/(m+1)$, and conditioned on this event the question is
	$\mathrm{append}_x$ of the restricted question.
\end{proof}

\begin{lemma}[Degree zero]\label{lem:main-induction-degree-zero}
	\lean{MIPStarRE.LDT.MainInductionStep.simplifiedAnswerDegreeZero, MIPStarRE.LDT.MainInductionStep.degreeZeroPointAveragedMeasurement, MIPStarRE.LDT.MainInductionStep.degreeZero_localPointDistance_le, MIPStarRE.LDT.MainInductionStep.degreeZero_globalPointDistance_le, MIPStarRE.LDT.MainInductionStep.degreeZero_pointConsistency_with_average_le}
	\leanok
	\uses{def:main-induction-error, lem:local-to-global}
	If $d=0$, the measurement $G_{g_a}=\E_vA^v_a$, indexed by the constant
	polynomials $g_a$, satisfies
	$A^u_a\ot I\simeq_{\delta+\sqrt{8m\eps}}I\ot G_{g_a}$.
\end{lemma}
\begin{proof}
	\leanok
	\uses{prop:simeq-to-approx, prop:triangle-inequality-for-approx_delta}
	For an edge $(u,v)$ of the hypercube graph along the axis-parallel line
	$\ell$, the point--line test gives
	$A^u_a\ot I\approx_{2\eps}I\ot B^\ell_a\approx_{2\eps}A^v_a\ot I$, so the
	local distance is at most $8\eps$.  Lemma~\ref{lem:local-to-global},
	applied outcome by outcome, bounds the global distance
	$D=\E_{u,v}\sum_a\|(A^u_a-A^v_a)\ot I\ket\psi\|^2$ by $8m\eps$.  The
	inconsistency of $G$ with $A$ differs from the self-consistency defect
	$\delta$ by at most $\sqrt D$ by Cauchy--Schwarz.
\end{proof}

\begin{lemma}[Dimension one]\label{lem:main-induction-dimension-one}
	\lean{MIPStarRE.LDT.MainInductionStep.simplifiedAnswerDimensionOneAxisMeasurement, MIPStarRE.LDT.MainInductionStep.simplifiedAnswerPositiveDegreeBase}
	\leanok
	\uses{def:main-induction-error}
	If $m=1$ and $d\ge1$, the measurement $G=B^\ell$ for the single
	axis-parallel line $\ell=\F_q$ satisfies
	$A^u_a\ot I\simeq_\eps I\ot G_{[g(u)=a]}$, and $\eps\le\sigma_{1,d}$ when
	$\eps\le1$.
\end{lemma}
\begin{proof}
	\leanok
	This is the axis-parallel test, and $\eps\le\eps^{1/32}$ for $\eps\le1$.
\end{proof}

\begin{lemma}[Large parameters]\label{lem:main-induction-large-error}
	\lean{MIPStarRE.LDT.MainInductionStep.one_le_simplifiedMainInductionError_of_large_parameter, MIPStarRE.LDT.MainInductionStep.simplifiedAnswerInductionOfOneLeError}
	\leanok
	\uses{def:main-induction-error}
	If $d\ge1$ and one of $\eps,\delta,d/q$ is at least one, then
	$\sigma_{m,d}(\eps,\delta,\gamma)\ge1$, and any measurement supported on a
	single polynomial satisfies the conclusion of
	Theorem~\ref{thm:main-induction}.
\end{lemma}
\begin{proof}
	\leanok
	The coefficient $1000d^2m^4$ is at least one, and consistency errors on a
	normalized state are at most one.
\end{proof}

\begin{lemma}[Compression]\label{lem:compression}
	\lean{MIPStarRE.LDT.MainInductionStep.compressed_postprocessed_consistency, MIPStarRE.LDT.MainInductionStep.compressed_answer_polynomial_consistency}
	\leanok
	\uses{lem:simultaneous-dilation-compression}
	Let $J$ be the isometric embedding of
	Theorem~\ref{thm:self-improvement-in-induction-section} and
	$\ket{\psi'}=(J\ot J)\ket\psi$.  If a measurement $T$ on the dilated space
	satisfies $(A^u_a\ot I_{\mathcal K})\ot I\simeq_\sigma I\ot T_{[h(u)=a]}$ on
	$\ket{\psi'}$, then $\widehat T_h=J^\dagger T_hJ$ is a measurement on the
	original space with $A^u_a\ot I\simeq_\sigma I\ot\widehat T_{[h(u)=a]}$.
\end{lemma}
\begin{proof}
	\leanok
	$\bra\psi A^u_a\ot\widehat T_h\ket\psi
		=\bra{\psi'}(A^u_a\ot I_{\mathcal K})\ot T_h\ket{\psi'}$, and
	$\sum_h\widehat T_h=J^\dagger J=I$.
\end{proof}

\begin{lemma}[Successor scalar recurrence]\label{lem:main-induction-recurrence}
	\lean{MIPStarRE.LDT.MainInductionStep.answerSuccessorAverageInductionError_le, MIPStarRE.LDT.MainInductionStep.answerSuccessorAverageDilationError_le, MIPStarRE.LDT.MainInductionStep.simplified_successor_coefficient_absorption, MIPStarRE.LDT.MainInductionStep.answerSuccessorPastingError_le_inductionError}
	\leanok
	\uses{def:main-induction-error, lem:restricted-probabilities, thm:ld-pasting-in-induction-section}
	Let $d\ge1$, $\eps,\delta,d/q\le1$, $S=\eps^{1/32}+\delta^{1/32}+(d/q)^{1/32}+\gamma^{1/8}$
	and $c=(m+1)/m$.  If the slices have average induction error
	$\bar\sigma\le1000d^2m^4c^{1/8}S$ and average self-improvement error
	$\bar\zeta\le3000(m+1)S$, then
	\[
		c(\bar\sigma+\bar\zeta)+21d^2(m+1)^2\sqrt m\,s\le1000d^2(m+1)^4S,
	\]
	where $s$ is the pasting parameter with $\zeta=\bar\zeta$.
\end{lemma}
\begin{proof}
	\leanok
	By concavity and Lemma~\ref{lem:restricted-probabilities}, the average
	induction error is at most $1000d^2m^4c^{1/8}S$, and
	$\bar\zeta=\E_x200(m+1)(\sqrt{\eps_x}+\sqrt{\delta_x}+\sqrt{d/q})$ is at
	most $3000(m+1)S$.  Subadditivity and $3000^{1/8}<3$ give
	$s\le4(m+1)^{1/8}S$.  With $c^{1/8}\le1+1/(8m)$ and $(m+1)^{1/8}\le2\sqrt m$,
	the gap between $1000d^2(m+1)^4$ and $1000d^2m^4c^{1+1/8}$ is at least
	$1000d^2(m+1)(\frac{23}8m^2+3m+1)$, which absorbs the two fresh terms.
\end{proof}

\begin{lemma}[Main induction for relaxed diagonal answers]\label{lem:answer-main-induction}
	\lean{MIPStarRE.LDT.MainInductionStep.simplifiedAnswerMainInduction, MIPStarRE.LDT.MainInductionStep.SimplifiedAnswerInductionConclusion, MIPStarRE.LDT.MainInductionStep.SimplifiedAnswerInductionHypothesis, MIPStarRE.LDT.MainInductionStep.SimplifiedAnswerSuccessorDilationData.ofInductionData}
	\leanok
	\uses{def:main-induction-error, def:restricted-strategy}
	The conclusion of Theorem~\ref{thm:main-induction} holds for every
	$(\eps,\delta,\gamma)$-good symmetric strategy whose diagonal-line
	measurements have all functions on the line as outcomes.
\end{lemma}
\begin{proof}
	\leanok
	\uses{lem:main-induction-degree-zero, lem:main-induction-dimension-one, lem:main-induction-large-error, lem:restricted-probabilities, thm:self-improvement-in-induction-section, thm:ld-pasting-in-induction-section, lem:compression, lem:main-induction-recurrence}
	The class of strategies with the relaxed diagonal answer alphabet is
	closed under restriction (Definition~\ref{def:restricted-strategy}),
	which is why the induction is carried out in this class.

	The case $d=0$ is Lemma~\ref{lem:main-induction-degree-zero}, and
	Lemma~\ref{lem:main-induction-large-error} handles $d\ge1$ with one of
	$\eps,\delta,d/q$ at least one.  Otherwise induct on $m$; the base case
	$m=1$ is Lemma~\ref{lem:main-induction-dimension-one}.  For the step from
	$m$ to $m+1$, apply the induction hypothesis to every $x$-restricted
	strategy, obtaining measurements $G_x$ with errors
	$\sigma_{m,d}(\eps_x,\delta_x,\gamma_x)$.  Apply
	Theorem~\ref{thm:self-improvement-in-induction-section} to all slices
	with a common ancilla and embedding $J$, extend the ambient strategy by
	the identity on the ancilla, and use the state $(J\ot J)\ket\psi$.  The
	projective output slices satisfy the hypotheses of
	Theorem~\ref{thm:ld-pasting-in-induction-section} with
	$\kappa=\bar\sigma+\bar\zeta$ and $\zeta=\bar\zeta$; boundedness uses
	permutation invariance to exchange the tensor factors.  Pasting gives a
	measurement on the dilated space, Lemma~\ref{lem:compression} returns it
	to the original space with the same error, and
	Lemma~\ref{lem:main-induction-recurrence} bounds the error by
	$\sigma_{m+1,d}(\eps,\delta,\gamma)$.
\end{proof}
