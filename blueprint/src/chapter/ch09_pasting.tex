\chapter{Pasting}
\label{chap:pasting}
\label{sec:ld-pasting}

A single positive contraction controls a whole sequence of independent
random slice measurements.  Its powers give both the cost of moving a
successful outcome to the start and the change in marginal success
probabilities.  Interpolation and a first-moment bound then finish the
argument.

The goal of this chapter is Theorem~\ref{thm:ld-pasting} at its end: for
every $k\ge d+1$, the slice measurements can be pasted into a global
measurement whose inconsistency with $A$ is
$\frac k{k-d}\kappa+21k^2\sqrt m\,s$, where $\kappa$ is the incompleteness
of the slices and $s$ collects the eighth roots of the small parameters.

Throughout this chapter $m,d\ge1$, $(\psi,A,B,L)$ is an
$(\eps,\delta,\gamma)$-good symmetric strategy for the $(m+1,q,d)$ test,
and the projective slice sub-measurements $\{G^x\}$ satisfy the hypotheses
of Definition~\ref{def:slice-family-hypotheses} with parameter $\zeta$.  We
assume $\eps,\delta,\gamma,\zeta\le1$ and $d\le q$.  Write
\[
	\omega(X)=\bra\psi X\ot I\ket\psi,\qquad
	G^x=\sum_gG^x_g,\qquad G=\E_xG^x,\qquad
	p=\omega(G),\qquad\kappa_0=1-p\le\kappa,
\]
and let $B^u$ be the line measurement in the last coordinate, as in
Notation~\ref{not:conditioned-on-last-direction}.

\begin{lemma}[Slice--line consistency]\label{lem:ld-gbcon}
	\lean{MIPStarRE.LDT.Pasting.pointVerticalLineSdd_of_axis_self, MIPStarRE.LDT.Pasting.ldGbcon_of_axis_self}
	\leanok
	\uses{def:slice-family-hypotheses, not:conditioned-on-last-direction}
	With $D=8m\eps+4\delta$ and $b=\zeta+\sqrt D$,
	\[
		I\ot A^{u,x}_a\approx_DI\ot B^u_{[f(x)=a]},\qquad
		G^x_{[g(u)=a]}\ot I\simeq_bI\ot B^u_{[f(x)=a]}.
	\]
\end{lemma}
\begin{proof}
	\leanok
	\uses{prop:simeq-to-approx, prop:triangle-inequality-for-approx_delta, prop:triangle-sub}
	Conditioning the axis-parallel test on the last direction costs at most
	$(m+1)\eps\le2m\eps$.  Point self-consistency and
	Propositions~\ref{prop:simeq-to-approx}
	and~\ref{prop:triangle-inequality-for-approx_delta} give the first
	relation, and Proposition~\ref{prop:triangle-sub} transfers the
	consistency hypothesis along it.
\end{proof}

\section{A positive contraction for random measurement words}
\label{sec:hat-consistency}

Set $Q_x=I-G^x$ and complete each slice measurement by $P^x_g=G^x_g$ and
$P^x_\bot=Q_x$.  Put $\Pi_x=\sum_aP^x_a\ot P^x_a$ and $K=\E_x\Pi_x$, so
$0\le K\le I$.

\begin{lemma}[Completion]\label{lem:pasting-completion}
	\lean{MIPStarRE.LDT.Pasting.gHatSelfConsistency_of_stronglySelfConsistent, MIPStarRE.LDT.Pasting.incompletePart_selfConsistency}
	\leanok
	\uses{def:slice-family-hypotheses, def:measurement-completion}
	The completed slice measurements satisfy
	\[
		\E_x\sum_a\|(P^x_a\ot I-I\ot P^x_a)\ket\psi\|^2
		=2\bra\psi(I-K)\ket\psi\le\alpha:=2\zeta.
	\]
\end{lemma}
\begin{proof}
	\leanok
	For fixed $x$ let
	$e_x=\sum_g\|(G^x_g\ot I-I\ot G^x_g)\ket\psi\|^2$.  Since
	$G^x\ot G^x\ge\sum_gG^x_g\ot G^x_g$, the mirror error of $G^x$ is at most
	$e_x$, and the mirror difference of $Q_x$ is the negative of that of
	$G^x$.  Hence the completed error is at most $2e_x$, and for a complete
	projective measurement it equals $2\bra\psi(I-\Pi_x)\ket\psi$.
\end{proof}

For independent uniform questions $y_1,\ldots,y_j$ and outcomes
$a_1,\ldots,a_j$ write $W=P^{y_j}_{a_j}\cdots P^{y_1}_{a_1}$, and let
$\E\sum_W$ average the questions and sum the outcomes.  Successive
outcome summation gives $\sum_WW^\dagger W=\sum_WWW^\dagger=I$, and
independence gives $\E\sum_WW\ot W=K^j$.

\begin{lemma}[Random-word mirror energy]\label{lem:random-word-energy}
	\lean{MIPStarRE.LDT.Pasting.matchingWordCorrelationAverage_eq_pow, MIPStarRE.LDT.Pasting.matchingOutcomeAverage_power_bound, MIPStarRE.LDT.Pasting.randomWordMirrorEnergy_eq, MIPStarRE.LDT.Pasting.randomWordMirrorEnergy_le, MIPStarRE.LDT.Pasting.ev_randomWordMirrorEnergy_le}
	\leanok
	\uses{lem:pasting-completion}
	With $M_W=W\ot I-I\ot W^\dagger$,
	\[
		\E\sum_WM_W^\dagger M_W=2(I-K^j)\le2j(I-K),
		\qquad
		\E\sum_W\|M_W\ket\psi\|^2\le j\alpha.
	\]
\end{lemma}
\begin{proof}
	\leanok
	Both diagonal terms sum to $I$, the cross terms average to $K^j$, and
	$I-K^j\le j(I-K)$ for $0\le K\le I$.  The second bound follows from
	Lemma~\ref{lem:pasting-completion}.
\end{proof}

\begin{lemma}[Stability of marginal probabilities]\label{lem:sequential-bot-marginal}
	\lean{MIPStarRE.LDT.Pasting.averagedSandwichMass_gap_le_sqrt, MIPStarRE.LDT.Pasting.completedWord_successProbability_ge_mean_sub_sqrt}
	\leanok
	\uses{lem:random-word-energy}
	Let $\rho$ be Alice's reduced state and
	$\Phi(\rho)=\E_x\sum_aP^x_a\rho P^x_a$.  For every effect $0\le M\le I$
	and $j\ge0$,
	$|\tr(M\Phi^j(\rho))-\tr(M\rho)|\le\sqrt{j\alpha}$.  In particular the
	$(j+1)$-st slice measurement of a random sequence succeeds with
	probability at least $p-\sqrt{j\alpha}$.
\end{lemma}
\begin{proof}
	\leanok
	In the weighted direct sum over words let $U_W=(W\ot I)\psi$ and
	$V_W=(I\ot W^\dagger)\psi$.  Both have norm one, and their $M\ot I$
	expectations are $\tr(M\Phi^j(\rho))$ and $\tr(M\rho)$.  Replacing $M$ by
	$M-I/2$ preserves the difference, and Cauchy--Schwarz with
	Lemma~\ref{lem:random-word-energy} bounds it by
	$\frac12\|U-V\|(\|U\|+\|V\|)\le\sqrt{j\alpha}$.
\end{proof}

\begin{lemma}[Moving a successful outcome]\label{lem:commute-g-half-sandwich}
	\lean{MIPStarRE.LDT.Pasting.ev_meanFineSliceEnergy_le_root_errors, MIPStarRE.LDT.Pasting.randomWordCommutatorEnergy_le_energy_roots, MIPStarRE.LDT.Pasting.randomWordCommutatorEnergy_le_coarse}
	\leanok
	\uses{lem:random-word-energy, thm:com-main}
	Let $\chi=30m(\gamma^{1/4}+\zeta^{1/4}+(d/q)^{1/4})$ be the error of
	Theorem~\ref{thm:com-main}.  For a length-$j$ random word $W$ and an
	independent uniform $x$,
	\[
		D_j=\E_x\E\sum_W\sum_g\|([G^x_g,W]\ot I)\psi\|^2
		\le2j(\sqrt\chi+4\sqrt\zeta).
	\]
\end{lemma}
\begin{proof}
	\leanok
	Write $P=G^x_g\ot I$ and $\Delta^y_h=G^y_h\ot I-I\ot G^y_h$.  The energy
	$\beta=\E_x\sum_g\bra\psi P(I-K)P\ket\psi$ satisfies
	$\sqrt\beta\le\sqrt\chi+\sqrt\zeta$, because
	$\sum_h(\Delta^y_h)^2\ge I-\Pi_y$ and
	$\Delta^y_hP\psi=([G^y_h,G^x_g]\ot I)\psi+P\Delta^y_h\psi$.  Since $P$
	commutes with $I\ot W^\dagger$,
	$([G^x_g,W]\ot I)\psi=PM_W\psi-M_WP\psi$, and
	Lemma~\ref{lem:random-word-energy} gives
	$\sqrt{D_j}\le\sqrt{2j}(\sqrt\zeta+\sqrt\beta)$.  Finally
	$\beta\le\min\{1,(\sqrt\chi+\sqrt\zeta)^2\}$ and $\zeta\le1$ give
	$(\sqrt\zeta+\sqrt\beta)^2\le\sqrt\chi+4\sqrt\zeta$.
\end{proof}

\section{Interpolation and consistency}
\label{sec:ld-sandwiching}

For independent uniform $x_1,\ldots,x_k$ and completed outcomes
$a_1,\ldots,a_k$ the sequential effect
$E^x_a=P^{x_1}_{a_1}\cdots P^{x_k}_{a_k}P^{x_k}_{a_k}\cdots P^{x_1}_{a_1}$
is positive and $\sum_aE^x_a=I$.

\begin{definition}[Distinct-success interpolation]\label{def:first-success-pasting}
	\lean{MIPStarRE.LDT.Pasting.HasDistinctSuccessSupport, MIPStarRE.LDT.Pasting.distinctSuccessInterpolant, MIPStarRE.LDT.Pasting.distinctSuccessPastedSubMeas, MIPStarRE.LDT.Pasting.distinctSuccessPastedMeasurement, MIPStarRE.LDT.Pasting.marginalRootBudget}
	\leanok
	\uses{def:polymeasurements, def:post-processing}
	If the successful indices of $(x,a)$ contain $d+1$ indices
	$j_1,\ldots,j_{d+1}$ with distinct coordinates $x_{j_\ell}$, select such a
	set canonically and output the Lagrange interpolant
	\[
		h_{x,a}(u,t)=\sum_{\ell=1}^{d+1}a_{j_\ell}(u)
		\prod_{v\ne\ell}\frac{t-x_{j_v}}{x_{j_\ell}-x_{j_v}},
	\]
	which has individual degree at most $d$ and restricts to $a_{j_\ell}$ on
	the slice $x_{j_\ell}$.  Otherwise output nothing.  The pasted
	sub-measurement is $T_{k,h}=\E_x\sum_{a:h_{x,a}=h}E^x_a$, with total
	$T_k$; the pasted measurement completes it at a fixed polynomial.  Set
	$L_k=\sum_{j=0}^{k-1}\sqrt j\le k(k-1)/2$.
\end{definition}

\begin{lemma}[Consistency]\label{lem:pasting-consistency}
	\lean{MIPStarRE.LDT.Pasting.ldSandwichLineOnePoint_simplified_coarse, MIPStarRE.LDT.Pasting.distinctSuccessPasted_verticalLine_consistency_coarse, MIPStarRE.LDT.Pasting.pointConsistency_of_verticalLineConsistency, MIPStarRE.LDT.Pasting.distinctSuccessPasted_pointConsistency_coarse}
	\leanok
	\uses{def:first-success-pasting, lem:ld-gbcon, lem:commute-g-half-sandwich}
	The inconsistency $\eta_k$ of $\{T_{k,h}\}$ with $A$ satisfies
	\[
		\eta_k\le2kb+2k(k-1)(\sqrt\chi+4\sqrt\zeta)+\sqrt D.
	\]
\end{lemma}
\begin{proof}
	\leanok
	\uses{prop:triangle-sub}
	Fix a position $i$, sum out later outcomes, and write
	$W=P^{x_{i-1}}_{a_{i-1}}\cdots P^{x_1}_{a_1}$, $P=G^{x_i}_g$ and
	$C=I-B^u_{[f(x_i)=g(u)]}$.  Compare $v=(PW\ot C^{1/2})\psi$ with
	$v'=(WP\ot C^{1/2})\psi$ in the direct sum over $u,x,a_{<i},g$.  The
	squared norm of $v$ is the probability $S_i$ of a successful mismatch at
	position $i$, $\|v-v'\|\le\sqrt{D_{i-1}}$ by
	Lemma~\ref{lem:commute-g-half-sandwich}, and $\|v'\|^2\le b$ by
	Lemma~\ref{lem:ld-gbcon}.  Hence $S_i\le2b+2D_{i-1}$.  If Bob's line
	polynomial differs from $h_{x,a}(u,\cdot)$, it disagrees at one of the
	selected coordinates, so a union bound over positions bounds the line
	inconsistency by $\sum_iS_i$.  Evaluation at a uniform point and
	Proposition~\ref{prop:triangle-sub} add $\sqrt D$.
\end{proof}

\section{Completeness from the expected number of successes}

\begin{lemma}[Completeness]\label{lem:first-success-completeness}
	\lean{MIPStarRE.LDT.Pasting.threshold_probability_ge_first_moment, MIPStarRE.LDT.Pasting.uniformQuestionCollision_le, MIPStarRE.LDT.Pasting.distinctSuccessPastedMass_ge_first_moment, MIPStarRE.LDT.Pasting.distinctSuccessPastedMass_ge_rootBudget}
	\leanok
	\uses{def:first-success-pasting, lem:sequential-bot-marginal}
	For $k\ge d+1$,
	\[
		\omega(T_k)\ge1-\frac{k}{k-d}\kappa_0-\frac{\sqrt\alpha\,L_k}{k-d}
		-\frac{k^2}q.
	\]
\end{lemma}
\begin{proof}
	\leanok
	Let $N$ count the successful outcomes.  By
	Lemma~\ref{lem:sequential-bot-marginal}, $\E N\ge kp-\sqrt\alpha L_k$.
	The pointwise inequality $N\le d+(k-d)\mathbf 1[N\ge d+1]$ gives
	$\Pr[N\ge d+1]\ge(\E N-d)/(k-d)$.  When the $k$ coordinates are distinct,
	$d+1$ successes suffice for interpolation, and a union bound shows that
	the coordinates collide with probability at most $k^2/q$.  The
	simplified exposition uses the sharper collision bound
	$\binom k2/q$; the formalization keeps $k^2/q$, which is absorbed in the
	final estimate.
\end{proof}

\begin{lemma}[Explicit error of the pasted measurement]\label{lem:pasting-explicit-error}
	\lean{MIPStarRE.LDT.Pasting.pointConsistency_completeAtOutcome, MIPStarRE.LDT.Pasting.distinctSuccessPastedMeasurement_pointConsistency, MIPStarRE.LDT.Pasting.distinctSuccessPastingError, MIPStarRE.LDT.Pasting.distinctSuccessPastingError_le_paperBound}
	\leanok
	\uses{def:first-success-pasting, lem:pasting-consistency, lem:first-success-completeness}
	Let $d\ge1$ and $k\ge d+1$, and assume $\eps,\delta,\zeta\le1$ and
	$d\le q$.  The pasted measurement, completed at a fixed polynomial, is
	consistent with $A$ with error
	\[
		\frac{k}{k-d}\kappa+2kb+2k(k-1)(\sqrt\chi+4\sqrt\zeta)+\sqrt D
		+\frac{\sqrt{2\zeta}\,L_k}{k-d}+\frac{k^2}q
		\le\frac{k}{k-d}\kappa+21k^2\sqrt m\,s.
	\]
\end{lemma}
\begin{proof}
	\leanok
	Completing a sub-measurement at a fixed outcome adds at most its missing
	mass to the inconsistency, so the error is at most
	$\eta_k+1-\omega(T_k)$, and the two preceding lemmas give the displayed
	sum.  Subadditivity of the square root and the standing assumptions give
	$\sqrt\chi+4\sqrt\zeta\le(\sqrt{30}+4)\sqrt m\,s$ and
	$b,\sqrt D\le3\sqrt m\,s$, and the last two terms are at most
	$k^2\sqrt m\,s$ each.  Collecting the coefficients gives at most
	$21k^2\sqrt m\,s$.
\end{proof}

\begin{theorem}[Pasting, quantitative form]\label{thm:ld-pasting}
	\lean{MIPStarRE.LDT.MainInductionStep.answerLdPastingSimplified, MIPStarRE.LDT.Pasting.simplifiedPastingPaperError, MIPStarRE.LDT.Pasting.pastingEighthSum}
	\leanok
	\uses{def:slice-family-hypotheses, def:polymeasurements}
	Let $m\ge1$ and $d\ge1$, and let $(\psi,A,B,L)$ be an
	$(\eps,\delta,\gamma)$-good symmetric strategy for the $(m+1,q,d)$ test.
	Let $\{G^x\}\subseteq\polysub mqd$ be projective slice sub-measurements
	satisfying the consistency, strong self-consistency, and boundedness
	hypotheses of Definition~\ref{def:slice-family-hypotheses} with
	parameter $\zeta$, and the completeness bound
	$\bra\psi G\ot I\ket\psi\ge1-\kappa$ for $G=\E_x\sum_gG^x_g$.  For every
	integer $k\ge d+1$ there is a measurement $H\in\polymeas{m+1}qd$ with
	\[
		A^u_a\ot I\simeq_\sigma I\ot H_{[h(u)=a]},\qquad
		\sigma=\frac{k}{k-d}\kappa+21k^2\sqrt m\,s,
	\]
	averaged over $u\in\F_q^{m+1}$, where
	$s=\eps^{1/8}+\delta^{1/8}+\gamma^{1/8}+\zeta^{1/8}+(d/q)^{1/8}$.
\end{theorem}
\begin{proof}
	\leanok
	\uses{lem:pasting-explicit-error, lem:pasting-completion, thm:com-main}
	If one of $\eps,\delta,\gamma,\zeta$ exceeds one or $d>q$, then $s\ge1$
	and a measurement supported on one polynomial works.  Otherwise
	Theorem~\ref{thm:com-main} and Lemma~\ref{lem:pasting-completion} supply
	the commutation and completed self-consistency inputs, and
	Lemma~\ref{lem:pasting-explicit-error} gives the bound.  The formalization
	states the theorem for the relaxed diagonal answer alphabet of
	Chapter~\ref{chap:induction}; commutativity of the point measurements is
	proved directly for that alphabet.
\end{proof}
