\chapter{Global variance of the point measurements}
\label{chap:variance}
\label{sec:variance}

Throughout this chapter, $(\psi,A,B,L)$ is an $(\eps,\delta,\gamma)$-good
symmetric strategy for the $(m,q,d)$ low individual degree test.

\begin{lemma}[Line answers and polynomial evaluation]\label{lem:generalize-b}
	\lean{MIPStarRE.LDT.GlobalVariance.generalizeBFromSchwartzZippel, MIPStarRE.LDT.GlobalVariance.generalizeBDeviationAtPolynomial_polysum_le_error}
	\leanok
	\uses{def:good-strategy, def:polymeasurements, def:approx_delta}
	Let $G\in\polysub mqd$.  On the axis-parallel lines test distribution,
	\[
		B^\ell_{[f(u)=g(u)]}\ot(G_g)^{1/2}\approx_{d/q}B^\ell_{g|_\ell}\ot(G_g)^{1/2}.
	\]
\end{lemma}
\begin{proof}
	\leanok
	\uses{cor:schwartz-zippel-individual}
	The squared distance is at most
	$\E_\ell\sum_g\sum_{f\ne g|_\ell}\bra\psi B^\ell_f\ot G_g\ket\psi\cdot
		\Prb_u[f(u)=g(u)]$, and two distinct univariate polynomials of degree at
	most $d$ agree at a uniform point of $\ell$ with probability at most $d/q$.
\end{proof}

\begin{lemma}[Local variance of the point measurements]\label{lem:local-variance-of-points}
	\lean{MIPStarRE.LDT.GlobalVariance.localVarianceDeviation_sum_le_localVarianceOfPointsError, MIPStarRE.LDT.GlobalVariance.localVarianceOfPointsFromEdgeDeviation}
	\leanok
	\uses{lem:generalize-b, def:hypercube-graph}
	Let $G\in\polysub mqd$.  On the edge distribution $(u,v)\sim C$,
	\[
		A^u_{g(u)}\ot(G_g)^{1/2}\approx_{24(\eps+\delta+d/q)}A^v_{g(v)}\ot(G_g)^{1/2};
	\]
	equivalently,
	$\sum_g\E_{(u,v)\sim C}\bra\psi(A^u_{g(u)}-A^v_{g(v)})^2\ot G_g\ket\psi
		\le24(\eps+\delta+d/q)$.
\end{lemma}
\begin{proof}
	\leanok
	\uses{prop:simeq-to-approx, prop:cab-approx-delta, prop:triangle-inequality-for-approx_delta}
	Let $\ell$ be the axis-parallel line through $u$ and $v$.  Six steps
	connect the two sides: point self-consistency moves $A^u_{g(u)}$ to the
	second tensor factor ($2\delta$), the point--line test moves it to
	$B^\ell_{[f(u)=g(u)]}$ ($2\eps$), Lemma~\ref{lem:generalize-b} replaces it
	by $B^\ell_{g|_\ell}$ ($d/q$), and the same three steps in reverse arrive at
	$A^v_{g(v)}$.  The triangle inequality for the state-dependent distance
	combines the six errors.
\end{proof}

\begin{lemma}[Global variance of the point measurements]\label{lem:global-variance-of-points}
	\lean{MIPStarRE.LDT.GlobalVariance.globalVarianceOfPointsFromLocalDeviation, MIPStarRE.LDT.GlobalVariance.globalVarianceDeviation_sum_le_m_mul_localVarianceDeviation_sum, MIPStarRE.LDT.GlobalVariance.globalVarianceDeviation_sum_le_of_localVarianceDeviation_sum_le}
	\leanok
	\uses{lem:local-variance-of-points, lem:local-to-global}
	Let $G\in\polysub mqd$.  For independent uniform $u,v\in\F_q^m$,
	\[
		A^u_{g(u)}\ot(G_g)^{1/2}\approx_{24m(\eps+\delta+d/q)}A^v_{g(v)}\ot(G_g)^{1/2}.
	\]
\end{lemma}
\begin{proof}
	\leanok
	Apply Lemma~\ref{lem:local-to-global} to the vector-valued function
	$X(u)=\bigoplus_g(A^u_{g(u)}\ot\sqrt{G_g})\ket\psi$ and use
	Lemma~\ref{lem:local-variance-of-points}.
\end{proof}
